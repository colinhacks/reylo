## Supported drivers

Reylo has no driver of its own: it runs each query on your app's database client, from any of the drivers below.

### Postgres

To run an imported Reylo query, pass a database connection/pool into the `.run()` method along with any parameters.

```ts twoslash tab="pg" dialect="postgres" tab-group="postgres-driver"
import pg from "pg";
import getUser from "./get_user.sql";

const db = new pg.Pool({ connectionString: process.env.DATABASE_URL });

const user = await getUser.run(db, { id: 1 });
```

```ts twoslash tab="postgres.js" dialect="postgres"
import postgres from "postgres";
import getUser from "./get_user.sql";

const db = postgres(process.env.DATABASE_URL);

const user = await getUser.run(db, { id: 1 });
```

```ts twoslash tab="Bun.sql" dialect="postgres"
import { SQL } from "bun";
import getUser from "./get_user.sql";

const db = new SQL(process.env.DATABASE_URL);

const user = await getUser.run(db, { id: 1 });
```

```ts twoslash tab="PGlite" dialect="postgres"
import { PGlite } from "@electric-sql/pglite";
import getUser from "./get_user.sql";

const db = new PGlite("./pgdata");

const user = await getUser.run(db, { id: 1 });
```

```ts twoslash tab="Neon" dialect="postgres"

import getUser from "./get_user.sql";

const db = neon(process.env.DATABASE_URL);

const user = await getUser.run(db, { id: 1 });
```

```ts twoslash tab="pg-promise" dialect="postgres"

const db = pgp(process.env.DATABASE_URL);

```

```ts twoslash tab="@vercel/postgres" dialect="postgres"

```


Transactions are also supported.

```ts twoslash tab="pg" dialect="postgres" tab-group="postgres-driver"
import pg from "pg";
import getUser from "./get_user.sql";
const db = new pg.Pool({ connectionString: process.env.DATABASE_URL });
// ---cut---
const client = await db.connect();
try {
  await client.query("begin");
  await getUser.run(client, { id: 1 });
  await client.query("commit");
} catch (error) {
  await client.query("rollback");
  throw error;
} finally {
  client.release();
}
```

```ts twoslash tab="postgres.js" dialect="postgres"
import postgres from "postgres";
import getUser from "./get_user.sql";
const db = postgres(process.env.DATABASE_URL);
// ---cut---
await db.begin((tx) => getUser.run(tx, { id: 1 }));
```

```ts twoslash tab="Bun.sql" dialect="postgres"
import { SQL } from "bun";
import getUser from "./get_user.sql";
const db = new SQL(process.env.DATABASE_URL);
// ---cut---

```

```ts twoslash tab="PGlite" dialect="postgres"
import { PGlite } from "@electric-sql/pglite";
import getUser from "./get_user.sql";
const db = new PGlite("./pgdata");
// ---cut---
await db.transaction((tx) => getUser.run(tx, { id: 1 }));
```

```ts twoslash tab="Neon" dialect="postgres"
import getUser from "./get_user.sql";
// ---cut---
import { Pool } from "@neondatabase/serverless";

// a transaction takes a Pool, over a WebSocket
const pool = new Pool({ connectionString: process.env.DATABASE_URL });
const client = await pool.connect();
try {
  await client.query("begin");
  await getUser.run(client, { id: 1 });
  await client.query("commit");
} catch (error) {
  await client.query("rollback");
  throw error;
} finally {
  client.release();
}
```

```ts twoslash tab="pg-promise" dialect="postgres"
import pgPromise from "pg-promise";

const pgp = pgPromise();
const db = pgp(process.env.DATABASE_URL);
// ---cut---

```

```ts twoslash tab="@vercel/postgres" dialect="postgres"

```

```ts twoslash tab="Drizzle" dialect="postgres"

import getUser from "./get_user.sql";
const db = drizzle(process.env.DATABASE_URL!);
// ---cut---

```

### MySQL

To run an imported Reylo query, pass a database connection/pool into the `.run()` method along with any parameters.

```ts twoslash tab="mysql2" dialect="mysql" tab-group="mysql-driver"
import mysql from "mysql2/promise";
import getUser from "./get_user.sql";

const db = mysql.createPool(process.env.DATABASE_URL);

const user = await getUser.run(db, { id: 1 });
```

```ts twoslash tab="mariadb" dialect="mysql"
// mariadb takes only a mariadb:// URL, which Reylo reads as MySQL
import mariadb from "mariadb";
import getUser from "./get_user.sql";

const db = mariadb.createPool(process.env.DATABASE_URL);

const user = await getUser.run(db, { id: 1 });
```

```ts twoslash tab="PlanetScale" dialect="mysql"
// @planetscale/database 1.x: a 2.0 client is a type error
import { connect } from "@planetscale/database";
import getUser from "./get_user.sql";

const db = connect({ url: process.env.DATABASE_URL });

const user = await getUser.run(db, { id: 1 });
```

```ts twoslash tab="Bun.sql" dialect="mysql"
import { SQL } from "bun";
import getUser from "./get_user.sql";

const db = new SQL(process.env.DATABASE_URL);

const user = await getUser.run(db, { id: 1 });
```

```ts twoslash tab="Drizzle" dialect="mysql"

import getUser from "./get_user.sql";

const db = drizzle(process.env.DATABASE_URL!);
```

Transactions are also supported.

```ts twoslash tab="mysql2" dialect="mysql" tab-group="mysql-driver"
import mysql from "mysql2/promise";
import getUser from "./get_user.sql";
const db = mysql.createPool(process.env.DATABASE_URL);
// ---cut---
const conn = await db.getConnection();
try {
  await conn.beginTransaction();
  await getUser.run(conn, { id: 1 });
  await conn.commit();
} catch (error) {
  await conn.rollback();
  throw error;
} finally {
  conn.release();
}
```

```ts twoslash tab="mariadb" dialect="mysql"

  await conn.beginTransaction();
  await getUser.run(conn, { id: 1 });
  await conn.commit();

  await conn.rollback();
  throw error;
} finally {
  conn.release();
}
```


### SQLite

To run an imported Reylo query, pass a database connection/pool into the `.run()` method along with any parameters.

```ts twoslash tab="better-sqlite3" dialect="sqlite" tab-group="sqlite-driver"
import Database from "better-sqlite3";
import getUser from "./get_user.sql";

const db = new Database("app.db");

const user = await getUser.run(db, { id: 1 });
```

```ts twoslash tab="node:sqlite" dialect="sqlite"
import { DatabaseSync } from "node:sqlite";
import getUser from "./get_user.sql";

const db = new DatabaseSync("app.db");

const user = await getUser.run(db, { id: 1 });
```

```ts twoslash tab="Bun.sql" dialect="sqlite"
import { SQL } from "bun";
import getUser from "./get_user.sql";

const db = new SQL({ adapter: "sqlite", filename: "app.db" });

const user = await getUser.run(db, { id: 1 });
```

```ts twoslash tab="libSQL" dialect="sqlite"
import { createClient } from "@libsql/client";
import getUser from "./get_user.sql";

const db = createClient({ url: "file:app.db" });

const user = await getUser.run(db, { id: 1 });
```

```ts twoslash tab="Turso" dialect="sqlite"
import { connect } from "@tursodatabase/database";
import getUser from "./get_user.sql";

const db = await connect("app.db");

const user = await getUser.run(db, { id: 1 });
```

```ts twoslash tab="Cloudflare D1" dialect="sqlite"
import type { D1Database } from "@cloudflare/workers-types";
import getUser from "./get_user.sql";

export default {
  async fetch(request: Request, env: { DB: D1Database }) {
    const user = await getUser.run(env.DB, { id: 1 });
    return Response.json(user);
  },
};
```

```ts twoslash tab="Durable Objects" dialect="sqlite"
/// <reference types="@cloudflare/workers-types" />

import { DurableObject } from "cloudflare:workers";
import getUser from "./get_user.sql";

export class Board extends DurableObject {
  user(id: number) {
    return getUser.runSync(this.ctx.storage.sql, { id });
  }
}
```

```ts twoslash tab="Drizzle" dialect="sqlite"

const db = drizzle("app.db");
```

Transactions are also supported.

```ts twoslash tab="better-sqlite3" dialect="sqlite" tab-group="sqlite-driver"
import Database from "better-sqlite3";
import getUser from "./get_user.sql";
const db = new Database("app.db");
// ---cut---
// the function must not await, so it runs .runSync()
db.transaction(() => getUser.runSync(db, { id: 1 }))();
```


