## Supported drivers

Pass `.run()` a client from any of these drivers, or a Drizzle database over one. A transaction, a savepoint or a connection taken from a client is a client too, as each example shows.

The type check reads your schema through a driver of its own, so your app can use any of these.

### Postgres

```ts twoslash tab="pg" dialect="postgres"
import pg from "pg";
import getUser from "./get_user.sql";

const db = new pg.Pool({ connectionString: process.env.DATABASE_URL });

const user = await getUser.run(db, { id: 1 });

const client = await db.connect();
try {
  await client.query("begin");
  await getUser.run(client, { id: 1 });
  await client.query("commit");
} catch (error) {
  await client.query("rollback");
  throw error;
} finally {
  client.release();
}
```

### MySQL

```ts twoslash tab="mysql2" dialect="mysql"
import mysql from "mysql2/promise";
import getUser from "./get_user.sql";

const db = mysql.createPool(process.env.DATABASE_URL);

const user = await getUser.run(db, { id: 1 });

const conn = await db.getConnection();
try {
  await conn.beginTransaction();
  await getUser.run(conn, { id: 1 });
  await conn.commit();
} catch (error) {
  await conn.rollback();
  throw error;
} finally {
  conn.release();
}
```

### SQLite

```ts twoslash tab="better-sqlite3" dialect="sqlite"
import Database from "better-sqlite3";
import getUser from "./get_user.sql";

const db = new Database("app.db");

const user = await getUser.run(db, { id: 1 });

// the function must not await, so it runs .runSync()
db.transaction(() => getUser.runSync(db, { id: 1 }))();
```

```ts twoslash tab="node:sqlite" dialect="sqlite"
import { DatabaseSync } from "node:sqlite";

const db = new DatabaseSync("app.db");

const user = await getUser.run(db, { id: 1 });

db.exec("begin");
try {
  getUser.runSync(db, { id: 1 });
  db.exec("commit");
} catch (error) {
  db.exec("rollback");
  throw error;
}
```

```ts twoslash tab="Cloudflare D1" dialect="sqlite"
import type { D1Database } from "@cloudflare/workers-types";
import getUser from "./get_user.sql";

export default {
  async fetch(request: Request, env: { DB: D1Database }) {
    // D1 has no transaction a query can join: its batch() takes D1's own statements
    const user = await getUser.run(env.DB, { id: 1 });
    return Response.json(user);
  },
};
```

```ts twoslash tab="Durable Objects" dialect="sqlite"
/// <reference types="@cloudflare/workers-types" />
// ---cut---
import { DurableObject } from "cloudflare:workers";
import getUser from "./get_user.sql";

export class Board extends DurableObject {
  user(id: number) {
    return getUser.runSync(this.ctx.storage.sql, { id });
  }

  userInTransaction(id: number) {
    return this.ctx.storage.transactionSync(() =>
      getUser.runSync(this.ctx.storage.sql, { id }),
    );
  }
}
```
